\grecommentary{Canticum B. Mariæ Virginis, Luc. 1:46-55}

\gresetinitiallines{0}
\gregorioscore{magnificat/Magnificat-1D-p1}
\textit{Wielbij \textcolor{gregoriocolor}{\grecross} * duszo moja, Pana.}

\gregorioscore{magnificat/Magnificat-1D-p2}
\textit{I rozradował się duch mój: * w Bogu, Zbawicielu moim.}

\gregorioscore{magnificat/Magnificat-1D-p3}
\textit{Iż wejrzał na niskość służebnice swojéj: * albowiem oto odtąd błogosławioną mię zwać będą wszystkie narody.}

\gregorioscore{magnificat/Magnificat-1D-p4}
\textit{Albowiem uczynił mi wielkie rzeczy, który możny jest: * i święte imię jego.}

\gregorioscore{magnificat/Magnificat-1D-p5}
\textit{A miłosierdzie jego od narodu do narodów: * bojącym się jego.}

\gregorioscore{magnificat/Magnificat-1D-p6}
\textit{Uczynił moc ramieniem swojem: * rozproszył pyszne myśli serca ich.}

\gregorioscore{magnificat/Magnificat-1D-p7}
\textit{Złożył mocarze z stolice: * a podwyższył niskie.}

\gregorioscore{magnificat/Magnificat-1D-p8}
\textit{Łaknące napełnił dobrami: * a bogacze z niczem puścił.}

\gregorioscore{magnificat/Magnificat-1D-p9}
\textit{Przyjął Izraela, sługę swego: * wspomniawszy na miłosierdzie swoje.}

\gregorioscore{magnificat/Magnificat-1D-p10}
\textit{Jako mówił do ojców naszych: * Abrahamowi i nasieniu jego na wieki.}

\gregorioscore{magnificat/Magnificat-1D-p11}
\textit{Chwała Ojcu i Synowi, * i Duchowi Świętemu.}

\gregorioscore{magnificat/Magnificat-1D-p12}
\textit{Jak była na początku, teraz i zawsze, * i na wieki wieków. Amen.}
