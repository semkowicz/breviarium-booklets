\section{Hymnus}

\gresetinitiallines{1}
\gregorioscore{hymnus/jesu_corona_virginum-p1}

\vspace{6pt}
{\centering
\textit{
O Jezu, Ty Dziewic Korono,
Którego w swem łonie poczęła,\\
Jedyna Matka-Dziewica,
Wysłuchaj łaskawie błagania.
}

}
\vspace{6pt}

\gresetinitiallines{0}

\gregorioscore{hymnus/jesu_corona_virginum-p2}

\vspace{6pt}
{\centering
\textit{
Ty, który chodzisz pośród lilji,
Ozdobion chwałą Oblubieńca,\\
I otoczony gronem Dziewic,
Oblubienice hojnie darzysz.
}

}
\vspace{6pt}

\gregorioscore{hymnus/jesu_corona_virginum-p3}

\vspace{6pt}
{\centering
\textit{
Gdziekolwiek idziesz one wszędy,
Dążą za Tobą z pieśnią chwały,\\
Biegną, rozbrzmiewa niebo całe,
Najsłodszym hymnem ich miłości.
}

}
\vspace{6pt}

\gregorioscore{hymnus/jesu_corona_virginum-p4}

\vspace{6pt}
{\centering
\textit{
Ciebie pokornie prosim, Panie,
Daj zmysłom naszym łaskę Twoją,\\
By odtąd były nieświadome,
Wszelkich ran grzechu i zepsucia.
}

}
\vspace{6pt}

\gregorioscore{hymnus/jesu_corona_virginum-p5}

\vspace{6pt}
{\centering
\textit{
Chwała, moc, cześć i uwielbienie,
Bogu Ojcu wraz z Synem Jego,\\
Świętemu Pocieszycielowi,
Przez wszystkie wieki wieków.
Amen.
}

}
